Day 02 - Services & Tags

Ansible - Install & Enable Nginx
-name: Install and start Nginx on Pathnex
 hosts: all
 become: yes

 tasks:
   - name: Install nginx
     yum:
       name: nginx
       state: present

    - name: Enable nginx
      services:
         name: nginx
         state: started
         enable: yes

# Kubernetes - Deployent with 2 Replicas
 apiVersion: app/v1
 kind: Deployment
 metadata:
   name: pathnex-deployment
 spec:
   replicas: 2
   selectors:
     matchLables:
       app: pathnex-app
    template:
      metadata:
        lables:
          app: pathnex-app
      spec:
        containers:
          - name: app
	    image: nginx
	    ports:
              - containerPort: 80
